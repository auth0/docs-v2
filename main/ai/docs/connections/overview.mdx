With Auth0 connections, users can log in with a supported social or enterprise external provider and authorize access via [Token Vault](/docs/secure/call-apis-on-users-behalf/token-vault) to share their data with AI agents. AI agents can then call third-party APIs on the user's behalf to perform tasks like schedule a meeting in Google Calendar, create a Microsoft Word document, and more. To learn more, read the [Call third-party APIs on users' behalf](/ai/docs/intro/call-third-party-apis-on-users-behalf) concept doc.

export const SearchAndFilterGrid = ({ items = [], filters = [] }) => {
  const fuzzySearch = (needle, haystack) => {
    if (!needle || !haystack) return false;
    const needleLower = needle.toLowerCase();
    const haystackLower = haystack.toLowerCase();
    return haystackLower.includes(needleLower);
  };

  const [searchTerm, setSearchTerm] = useState("");
  const [filter, setFilter] = useState(null);
  const filterButtonsRef = useRef(null);
  const indicatorRef = useRef(null);

  const filteredItems = useMemo(() => {
    let filtered = items;

    if (searchTerm.trim()) {
      filtered = filtered.filter((item) => {
        return (
          fuzzySearch(searchTerm, item.title) ||
          fuzzySearch(searchTerm, item.description)
        );
      });
    }

    if (filter) {
      filtered = filtered.filter((item) => item.type === filter);
    }

    return filtered.sort((a, b) => a.title.localeCompare(b.title));
  }, [searchTerm, filter]);

  const searchClasses = [ "connection-card-search", 
    "inline-flex", "flex-row", "items-center", "text-sm", "mt-2",
    "border", "gap-2", "rounded-2xl"
  ].join(" ");
  const filterClasses = [ "connection-card-filters", "justify-end",
    "inline-flex", "flex-row", "items-center", "text-sm", "mt-2"
  ].join(" ");
  const buttonClasses = [ "px-4", "py-2", "border-y",
    "first:rounded-l-2xl", "last:rounded-r-2xl"
  ].join(" ");

  return (
  <>
  <Columns cols={2} className="mt-6">
    <label class={searchClasses}>
      <Icon icon="magnifying-glass" className="ml-4" />
      <input type="text" placeholder="Search..." value={searchTerm} class="grow py-2"
             onChange={(e) => setSearchTerm(e.target.value)} />
    </label>
    <div class={filterClasses} ref={filterButtonsRef}>
      <button onClick={() => setFilter(null)}
              class={buttonClasses.concat(filter === null ? " active" : "")}>
        All
      </button>
      {filters.map((f) => (
      <button onClick={() => setFilter(f)} key={f}
              class={buttonClasses.concat(filter === f ? " active" : "")}>
        {f}
      </button>
      ))}
    </div>

    {filteredItems.map((item) => {
    return (
      <Card title={item.title} icon={item.icon} href={item.href} className="connection-card">
        <p>{item.description}</p>
        <p><Badge color="purple">{item.type}</Badge></p>
      </Card>
    );
    })} 
  </Columns>

  {filteredItems.length === 0 && (
  <Tile title="No results found" className="my-8 connection-card-no-results"
        description={`No results for "${searchTerm}". Try adjusting your search or filter criteria.`}
  >
    <Icon icon="magnifying-glass" size={40} />
  </Tile>
  )}

  </>
  );
};
