```ts Example
export interface ResetPasswordMfaPushChallengePushMembers extends BaseMembers {
  screen: ScreenMembersOnResetPasswordMfaPushChallengePush;

  /**
   * Poursuit le défi de notification push
   * @param payload Options personnalisées facultatives à inclure avec la requête
   */
  continue(payload?: CustomOptions): Promise<void>;

  /**
   * Renvoie la notification push
   * @param payload Options personnalisées facultatives à inclure avec la requête
   */
  resendPushNotification(payload?: CustomOptions): Promise<void>;

  /**
   * Passe à la saisie manuelle du code de vérification
   * @param payload Options personnalisées facultatives à inclure avec la requête
   */
  enterCodeManually(payload?: CustomOptions): Promise<void>;

  /**
   * Permet d'essayer une autre méthode d'authentification
   * @param payload Options personnalisées facultatives à inclure avec la requête
   */
  tryAnotherMethod(payload?: CustomOptions): Promise<void>;

  /**
   * Permet d'interroger par sondage le défi de notification push jusqu'à son approbation.
   * @param options de type {@link MfaPollingOptions}
   */
  pollingManager(options?: MfaPollingOptions): MfaPushPollingControl;
}
```
