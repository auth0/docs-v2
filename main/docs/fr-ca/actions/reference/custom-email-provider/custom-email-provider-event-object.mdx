L’objet `event` du déclencheur Actions custom-email-provider fournit des renseignements contextuels sur son exécution.

```json Event Object
{
  "secrets": {},
  "client": {
    "client_id": "string",
    "metadata": {},
    "name": "string"
  },
  "notification": {
    "from": "string",
    "html": "string",
    "locale": "string",
    "message_type": "string",
    "subject": "string",
    "text": "string",
    "to": "string"
  },
  "request": {},
  "tenant": {
    "id": "string"
  },
  "user": {}
}
```

<div id="eventclient">
  ## `event.client`
</div>

<ResponseField name="event.client" type="object">
  Renseignements sur le client avec lequel cette transaction de connexion a été lancée.

  <Expandable title="propriétés d’event.client" defaultOpen>
    <ResponseField name="client_id" type="string">
      L’ID client de l’application à laquelle l’utilisateur se connecte.
    </ResponseField>

    <ResponseField name="metadata" type="dictionary">
      Un objet qui contient d’autres propriétés de l’application.
    </ResponseField>

    <ResponseField name="name" type="string">
      Le nom de l’application (tel que défini dans le Dashboard).
    </ResponseField>
  </Expandable>
</ResponseField>

<div id="eventconnection">
  ## `event.connection`
</div>

<ResponseField name="event.connection" type="object" post={["optional"]}>
  Informations sur la connexion utilisée pour authentifier l’utilisateur.

  <Expandable title="propriétés de event.connection" defaultOpen>
    <ResponseField name="id" type="string">
      L’identifiant unique de la connexion.
    </ResponseField>

    <ResponseField name="metadata" type="dictionary" post={["optional"]}>
      Métadonnées associées à la connexion.
    </ResponseField>

    <ResponseField name="name" type="string">
      Le nom de la connexion utilisée pour authentifier l’utilisateur (par exemple, `twitter` ou `some-g-suite-domain`).
    </ResponseField>

    <ResponseField name="strategy" type="string">
      Le type de connexion. Pour les connexions sociales, `event.connection.strategy === event.connection.name`. Pour les connexions d’entreprise, la stratégie peut être `waad` (Windows Azure AD), `ad` (Active Directory/LDAP), `auth0` (connexions à une base de données), etc.
    </ResponseField>
  </Expandable>
</ResponseField>

  ## `event.notification`
</div>

<ResponseField name="event.notification" type="object">
  <Expandable title="propriétés de event.notification" defaultOpen>
    <ResponseField name="from" type="string">
      Adresse courriel de l’expéditeur.
    </ResponseField>

    <ResponseField name="html" type="string">
      Modèle HTML généré.
    </ResponseField>

    <ResponseField name="locale" type="string">
      La locale utilisée pour générer le message, par exemple `en_US`, telle que définie dans la spécification BCP-47.
    </ResponseField>

    <ResponseField name="message_type" type="string">
      Le type de message envoyé, par exemple `verify_email` ou `welcome_email`.

      Valeurs autorisées : `verify_email`, `verify_email_by_code`, `reset_email`, `reset_email_by_code`, `welcome_email`, `verification_code`, `mfa_oob_code`, `enrollment_email`, `blocked_account`, `stolen_credentials`, `try_provider_configuration_email`, `organization_invitation`
    </ResponseField>

    <ResponseField name="subject" type="string">
      Objet du courriel.
    </ResponseField>

    <ResponseField name="text" type="string">
      Modèle de texte généré.
    </ResponseField>

    <ResponseField name="to" type="string">
      Adresse courriel du destinataire.
    </ResponseField>
  </Expandable>
</ResponseField>
