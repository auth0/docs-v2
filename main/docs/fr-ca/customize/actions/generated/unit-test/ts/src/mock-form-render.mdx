import type { Event, PostLoginAPI } from '@auth0/actions/post-login/v3';

const CONSENT_FORM_ID = 'marketing-consent-form';

/**
* Gestionnaire appelé lors de l'exécution d'un flux PostLogin.
* Affiche un Form personnalisé demandant le consentement marketing lorsque
* l'utilisateur n'y a pas encore répondu. `api.prompt.render` ne peut pas être
* combiné à `api.redirect.sendUserTo` dans une même exécution – choisissez-en
* un seul par action.
*
* @param {Event} event - Détails sur l'utilisateur et le contexte dans lequel il se connecte.
* @param {PostLoginAPI} api - Interface dont les méthodes permettent de modifier le comportement de la connexion.
*/
exports.onExecutePostLogin = async (event: Event, api: PostLoginAPI) => {
  const hasAnsweredConsent = typeof event.user.user_metadata.marketing_consent === 'boolean';

  if (hasAnsweredConsent) {
    return;
  }

  api.prompt.render(CONSENT_FORM_ID, {
    vars: { firstName: event.user.given_name ?? event.user.name },
  });
};

/**
* Gestionnaire appelé lorsque l'utilisateur soumet le Form affiché par
* onExecutePostLogin.
*
* @param {Event} event - Détails sur l'utilisateur et le contexte dans lequel il se connecte.
* @param {PostLoginAPI} api - Interface dont les méthodes permettent de modifier le comportement de la connexion.
*/
exports.onContinuePostLogin = async (event: Event, api: PostLoginAPI) => {
  const consent = event.prompt?.fields?.consent;

  if (typeof consent !== 'boolean') {
    api.access.deny('Marketing consent response was not submitted.');
    return;
  }

  api.user.setUserMetadata('marketing_consent', consent);
};
