```ts Example
export interface SignupOptions {
  email?: string;
  username?: string;
  phone?: string;
  phoneCountryCode?: string;
  captcha?: string;
  [key: string]: string | number | boolean | undefined;
}
```

<ParamField body="phoneCountryCode?" type="string">
  ユーザーが `phone` 用に選択した ISO 3166-1 alpha-2 の国コード (例: `'US'`) で、`countryCodes.available` 内の `code` から指定します。送信された国が優先されます。サーバーがその国番号を先頭に付加するため、`phone` には国番号を含まない国内番号を指定してください。省略した場合は既存の動作が維持され、サーバーが自動的に国を判定し、`pickCountryCode()` が別画面で選択を変更します。
</ParamField>
