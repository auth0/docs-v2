---
title: "ConfirmationMembers"
---

Confirmation画面のメンバーを表すインターフェース。

```ts Example
export interface ConfirmationMembers extends BaseMembers {
  /**
   * OTP検証後にアカウント作成を続行します。
   * @param payload リクエストに含める任意のカスタムオプション。
   */
  proceedToSignup(payload?: CustomOptions): Promise<void>;

  /**
   * 自動サインアップを拒否し、アカウントを作成せずにユーザーをログイン画面に戻します。
   * @param payload リクエストに含める任意のカスタムオプション。
   */
  goBack(payload?: CustomOptions): Promise<void>;
}
```
