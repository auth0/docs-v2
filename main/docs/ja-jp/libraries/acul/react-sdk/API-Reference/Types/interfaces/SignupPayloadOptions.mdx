```ts Example
export interface SignupPayloadOptions {
  email?: string;
  username?: string;
  phoneNumber?: string;
  phoneCountryCode?: string;
  password?: string;
  captcha?: string;
  [key: string]: string | number | boolean | undefined;
}
```

<ParamField body="phoneCountryCode?" type="string">
  ユーザーが`phoneNumber`用に選択した ISO 3166-1 alpha-2 の国 (例: `'US'`) 。`countryCodes.available`内の`code`から選択します。送信された国が優先されます。サーバーがその国のダイヤルコードを先頭に付加するため、`phoneNumber`にはダイヤルコードを含まない国内番号を指定してください。省略すると、サーバーが国を自動的に判定し、`pickCountryCode()`が別画面で選択を変更する既存の動作が維持されます。
</ParamField>
