```json Event Object
    "geoip": {},

    "id": "string"
  },
  "user": {}
}
```

<div id="eventnotification">
  ## `event.notification`
</div>

<ResponseField name="event.notification" type="object">
  <Expandable title="event.notification のプロパティ" defaultOpen>
    <ResponseField name="as_text" type="string">
      レンダリングされた、テキストメッセージとして配信可能なテキストです。
    </ResponseField>

    <ResponseField name="as_voice" type="string">
      レンダリングされた、音声テキストメッセージとして配信可能なテキストです。
    </ResponseField>

    <ResponseField name="code" type="string" post={["optional"]}>
      一部のタイプ (例: `otp_verify`、`otp_enroll`) でこのメッセージ用に生成されるワンタイムパスワードです。指定されている場合は、エンドユーザーに確実に伝達することが重要です。
    </ResponseField>

    <ResponseField name="delivery_method" type="string">
      メッセージの配信方法です。`text` または `voice` を指定できます。

      許可される値: `text`、`voice`
    </ResponseField>

    <ResponseField name="from" type="string" post={["optional"]}>
      送信者の E.164 準拠の電話番号です。
    </ResponseField>

    <ResponseField name="locale" type="string" post={["optional"]}>
      メッセージのレンダリングに使用した locale です。BCP-47 仕様で定義される `en_US` などが該当します。
    </ResponseField>

    <ResponseField name="message_type" type="string">
      送信されるメッセージのタイプです。`otp_verify` や `blocked_account` などがあります。

      許可される値: `otp_verify`、`otp_enroll`、`blocked_account`、`change_password`、`password_breach`
    </ResponseField>

    <ResponseField name="recipient" type="string">
      受信者の E.164 準拠の電話番号です。
    </ResponseField>
  </Expandable>
</ResponseField>
